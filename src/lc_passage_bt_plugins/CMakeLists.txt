cmake_minimum_required(VERSION 3.16)
project(lc_passage_bt_plugins)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(lc_passage_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
add_library(lc_passage_bt_guards SHARED src/bt_guards.cpp)
ament_target_dependencies(lc_passage_bt_guards rclcpp lc_passage_msgs nav_msgs behaviortree_cpp_v3)
target_compile_definitions(lc_passage_bt_guards PRIVATE BT_PLUGIN_EXPORT)
install(TARGETS lc_passage_bt_guards LIBRARY DESTINATION lib)
install(FILES LICENSE README.md DESTINATION share/${PROJECT_NAME})
ament_package()
