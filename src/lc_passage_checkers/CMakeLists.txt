cmake_minimum_required(VERSION 3.16)
project(lc_passage_checkers)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp rclcpp_lifecycle lc_passage_msgs nav2_core nav2_util
  nav2_costmap_2d pluginlib tf2 tf2_geometry_msgs)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(lc_passage_checkers SHARED src/checkers.cpp)
ament_target_dependencies(lc_passage_checkers ${dependencies})
pluginlib_export_plugin_description_file(nav2_core checkers_plugin.xml)
install(TARGETS lc_passage_checkers LIBRARY DESTINATION lib)
install(FILES LICENSE README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(passage_checker_load_test test/plugin_load_test.cpp)
  ament_target_dependencies(passage_checker_load_test nav2_core pluginlib)
endif()
ament_package()
