cmake_minimum_required(VERSION 3.16)
project(lc_passage_map)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp rclcpp_components lc_passage_common lc_passage_msgs
  nav_msgs sensor_msgs tf2 tf2_ros tf2_geometry_msgs visualization_msgs)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(passage_map_component SHARED src/passage_map_node.cpp)
target_include_directories(passage_map_component PRIVATE include)
ament_target_dependencies(passage_map_component ${dependencies})
rclcpp_components_register_node(passage_map_component
  PLUGIN "lc_passage_map::PassageMapNode" EXECUTABLE passage_map_node)
install(TARGETS passage_map_component LIBRARY DESTINATION lib)
install(FILES LICENSE README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(beam_filter_tests test/beam_filter_tests.cpp)
  target_include_directories(beam_filter_tests PRIVATE include)
  ament_target_dependencies(beam_filter_tests lc_passage_common)
  add_test(NAME passage_beam_filter COMMAND beam_filter_tests)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(passage_map_ros_test test/ros_integration_test.cpp TIMEOUT 60)
  target_include_directories(passage_map_ros_test PRIVATE include)
  ament_target_dependencies(passage_map_ros_test ${dependencies})
  target_compile_definitions(passage_map_ros_test PRIVATE
    PASSAGE_COMPONENT_LIBRARY="$<TARGET_FILE:passage_map_component>")
  add_dependencies(passage_map_ros_test passage_map_component)
endif()
ament_package()
