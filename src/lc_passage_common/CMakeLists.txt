cmake_minimum_required(VERSION 3.16)
project(lc_passage_common)
find_package(ament_cmake REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(nlohmann_json std_msgs geometry_msgs visualization_msgs)
if(BUILD_TESTING)
  enable_testing()
  add_executable(passage_config_tests test/config_tests.cpp)
  target_compile_features(passage_config_tests PRIVATE cxx_std_17)
  target_include_directories(passage_config_tests PRIVATE include)
  target_link_libraries(passage_config_tests nlohmann_json::nlohmann_json)
  add_test(NAME passage_config COMMAND passage_config_tests)
endif()
ament_package()
