cmake_minimum_required(VERSION 3.18)
project(passage_map_editor VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(QT NAMES Qt5 Qt6 REQUIRED COMPONENTS Core Gui Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Gui Widgets)
find_package(yaml-cpp REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
  set(EDITOR_YAML_TARGET yaml-cpp::yaml-cpp)
else()
  set(EDITOR_YAML_TARGET yaml-cpp)
endif()
# Deliberately no planner headers, target, source path, or solver dependency.
add_library(passage_editor_ui STATIC
  src/document.cpp src/document.hpp src/map_canvas.cpp src/map_canvas.hpp
  src/planner_client.cpp src/planner_client.hpp src/main_window.cpp src/main_window.hpp)
target_include_directories(passage_editor_ui PUBLIC src)
target_link_libraries(passage_editor_ui PUBLIC Qt${QT_VERSION_MAJOR}::Widgets ${EDITOR_YAML_TARGET})
target_compile_options(passage_editor_ui PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_executable(passage_map_editor src/main.cpp)
target_link_libraries(passage_map_editor PRIVATE passage_editor_ui)
option(EDITOR_BUILD_TESTS "Build native Qt interaction and protocol tests" ON)
if(EDITOR_BUILD_TESTS)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
  enable_testing()
  add_executable(passage_editor_tests tests/editor_tests.cpp)
  target_link_libraries(passage_editor_tests PRIVATE passage_editor_ui Qt${QT_VERSION_MAJOR}::Test)
  add_executable(editor_protocol_fixture tests/protocol_fixture.cpp)
  target_link_libraries(editor_protocol_fixture PRIVATE Qt${QT_VERSION_MAJOR}::Core)
  add_dependencies(passage_editor_tests editor_protocol_fixture)
  add_test(NAME editor_native COMMAND passage_editor_tests)
  set_tests_properties(editor_native PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(EDITOR_INTEGRATION_CLI)
    set_property(TEST editor_native APPEND PROPERTY ENVIRONMENT
      "TDT_PASSAGE_CLI=${EDITOR_INTEGRATION_CLI}"
      "TDT_EXAMPLES_DIR=${EDITOR_INTEGRATION_EXAMPLES}")
  endif()
endif()
install(TARGETS passage_map_editor RUNTIME DESTINATION bin)
